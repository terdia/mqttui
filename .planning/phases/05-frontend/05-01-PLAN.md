1. Create `mqttui/socketio_batch.py` with a BatchEmitter class:
   ```python
   import time
   import threading
   from flask_socketio import SocketIO

   class BatchEmitter:
       """Collects MQTT messages and emits them in batches every 100ms."""
       def __init__(self, socketio: SocketIO, interval_ms: int = 100):
           self._socketio = socketio
           self._interval = interval_ms / 1000.0
           self._buffer = []
           self._lock = threading.Lock()
           self._timer = None
           self._running = False

       def start(self):
           self._running = True
           self._schedule_flush()

       def stop(self):
           self._running = False
           if self._timer:
               self._timer.cancel()

       def enqueue(self, message: dict):
           with self._lock:
               self._buffer.append(message)

       def _schedule_flush(self):
           if not self._running:
               return
           self._timer = threading.Timer(self._interval, self._flush)
           self._timer.daemon = True
           self._timer.start()

       def _flush(self):
           with self._lock:
               batch = self._buffer
               self._buffer = []
           if batch:
               self._socketio.emit('mqtt_messages_batch', batch)
           self._schedule_flush()

   _batch_emitter = None

   def init_batch_emitter(socketio: SocketIO, interval_ms: int = 100):
       global _batch_emitter
       _batch_emitter = BatchEmitter(socketio, interval_ms)
       _batch_emitter.start()
       return _batch_emitter

   def get_batch_emitter():
       return _batch_emitter
   ```

3. Refactor `static/script.js` to Alpine.js component architecture:
   - Keep Vis.js network init and Chart.js init as standalone functions (per user decision: keep them as-is)
   - Create Alpine.js data functions:
     ```javascript
     const socket = io();

     // Alpine.js global store for shared state
     document.addEventListener('alpine:init', () => {
         Alpine.store('mqtt', {
             connected: false,
             selectedTopic: 'all',
             messageCount: 0,
         });
     });

     function mqttuiApp() {
         return {
             sidebarOpen: true,
             init() {
                 // Handle batch messages from server
                 socket.on('mqtt_messages_batch', (batch) => {
                     batch.forEach(msg => {
                         this.$dispatch('mqtt-message', msg);
                         Alpine.store('mqtt').messageCount++;
                         updateNetwork(msg);
                         updateTopicFilter(msg.topic);
                     });
                 });
                 // Fallback for single messages (backward compat)
                 socket.on('mqtt_message', (data) => {
                     this.$dispatch('mqtt-message', data);
                     Alpine.store('mqtt').messageCount++;
                     updateNetwork(data);
                     updateTopicFilter(data.topic);
                 });
             },
             toggleSidebar() { this.sidebarOpen = !this.sidebarOpen; },
         };
     }

     function messageListComponent() {
         return {
             messages: [],
             init() {
                 this.loadMessages();
                 window.addEventListener('mqtt-message', (e) => {
                     const msg = e.detail;
                     const filter = Alpine.store('mqtt').selectedTopic;
                     if (filter === 'all' || msg.topic === filter) {
                         this.messages.unshift(msg);
                         if (this.messages.length > 100) this.messages.pop();
                     }
                 });
             },
             async loadMessages() {
                 const params = new URLSearchParams({ limit: '50' });
                 const topic = Alpine.store('mqtt').selectedTopic;
                 if (topic && topic !== 'all') params.append('topic', topic);
                 const resp = await fetch(`/api/messages?${params}`);
                 const data = await resp.json();
                 this.messages = data.messages || [];
             },
             formatPayload(payload) {
                 try { return JSON.stringify(JSON.parse(payload), null, 2); }
                 catch { return payload; }
             },
             formatTime(ts) {
                 return new Date(ts).toLocaleTimeString();
             },
         };
     }

     function statsComponent() {
         return {
             connections: 0,
             topics: 0,
             messageTotal: 0,
             init() {
                 this.fetchStats();
                 setInterval(() => this.fetchStats(), 5000);
             },
             async fetchStats() {
                 const resp = await fetch('/stats');
                 const data = await resp.json();
                 this.connections = data.connection_count;
                 this.topics = data.topic_count;
                 this.messageTotal = data.message_count;
             },
         };
     }

     function publishComponent() {
         return {
             topic: '',
             message: '',
             async submit() {
                 await fetch('/publish', {
                     method: 'POST',
                     headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
                     body: `topic=${encodeURIComponent(this.topic)}&message=${encodeURIComponent(this.message)}`
                 });
                 this.topic = '';
                 this.message = '';
             },
         };
     }
     ```
